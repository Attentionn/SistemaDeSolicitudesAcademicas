## Procedimientos Almacenados

### 1. Procedimiento para Estadísticas de Solicitudes

```sql
-- Función para obtener estadísticas de solicitudes por estudiante
CREATE FUNCTION get_student_accommodation_stats(student_id INTEGER)
RETURNS TABLE (
    total_solicitudes INTEGER,
    solicitudes_aprobadas INTEGER,
    solicitudes_pendientes INTEGER,
    solicitudes_rechazadas INTEGER
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        COUNT(*) as total_solicitudes,
        COUNT(CASE WHEN status = 'approved' THEN 1 END) as solicitudes_aprobadas,
        COUNT(CASE WHEN status = 'pending' THEN 1 END) as solicitudes_pendientes,
        COUNT(CASE WHEN status = 'rejected' THEN 1 END) as solicitudes_rechazadas
    FROM Accommodations 
    WHERE studentId = student_id;
END;
$$ LANGUAGE plpgsql;
```

### 2. Procedimiento para Reportes por Período

```sql
-- Función para generar reportes por período
CREATE FUNCTION get_accommodation_report(
    start_date DATE,
    end_date DATE,
    report_type VARCHAR(20)
)
RETURNS TABLE (
    periodo VARCHAR(50),
    total_solicitudes INTEGER,
    aprobadas INTEGER,
    pendientes INTEGER,
    rechazadas INTEGER
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        CONCAT(DATE(start_date), ' - ', DATE(end_date)) as periodo,
        COUNT(*) as total_solicitudes,
        COUNT(CASE WHEN status = 'approved' THEN 1 END) as aprobadas,
        COUNT(CASE WHEN status = 'pending' THEN 1 END) as pendientes,
        COUNT(CASE WHEN status = 'rejected' THEN 1 END) as rechazadas
    FROM Accommodations 
    WHERE DATE(requestedDate) BETWEEN start_date AND end_date
    AND (report_type = 'all' OR type = report_type);
END;
$$ LANGUAGE plpgsql;
```

### 3. Procedimiento para Gestión de Faltas

```sql
-- Función para registrar falta con validaciones
CREATE FUNCTION register_absence(
    p_student_id INTEGER,
    p_course_id INTEGER,
    p_teacher_id INTEGER,
    p_fecha DATE,
    p_materia VARCHAR(255),
    p_motivo TEXT,
    p_tipo VARCHAR(20)
)
RETURNS INTEGER AS $$
DECLARE
    new_absence_id INTEGER;
BEGIN
    -- Validar que el estudiante existe
    IF NOT EXISTS (SELECT 1 FROM Users WHERE id = p_student_id AND role = 'student') THEN
        RAISE EXCEPTION 'Estudiante no encontrado';
    END IF;
    
    -- Validar que el curso existe
    IF NOT EXISTS (SELECT 1 FROM Courses WHERE id = p_course_id) THEN
        RAISE EXCEPTION 'Curso no encontrado';
    END IF;
    
    -- Validar que el profesor existe
    IF NOT EXISTS (SELECT 1 FROM Users WHERE id = p_teacher_id AND role = 'teacher') THEN
        RAISE EXCEPTION 'Profesor no encontrado';
    END IF;
    
    -- Insertar la falta
    INSERT INTO Absences (studentId, courseId, teacherId, fecha, materia, motivo, tipo)
    VALUES (p_student_id, p_course_id, p_teacher_id, p_fecha, p_materia, p_motivo, p_tipo)
    RETURNING id INTO new_absence_id;
    
    RETURN new_absence_id;
END;
$$ LANGUAGE plpgsql;
```

## Triggers y Funciones

### 1. Trigger para Actualización Automática de Timestamps

```sql
-- Trigger para actualizar updatedAt automáticamente
CREATE TRIGGER update_users_timestamp
    AFTER UPDATE ON Users
    FOR EACH ROW
    BEGIN
        UPDATE Users SET updatedAt = CURRENT_TIMESTAMP WHERE id = NEW.id;
    END;

CREATE TRIGGER update_courses_timestamp
    AFTER UPDATE ON Courses
    FOR EACH ROW
    BEGIN
        UPDATE Courses SET updatedAt = CURRENT_TIMESTAMP WHERE id = NEW.id;
    END;

CREATE TRIGGER update_accommodations_timestamp
```

### 2. Trigger para Notificaciones de Cambio de Estado

```sql
-- Función para notificar cambios de estado
CREATE FUNCTION notify_accommodation_status_change()
RETURNS TRIGGER AS $$
BEGIN
    -- Solo notificar si el estado cambió
    IF OLD.status != NEW.status THEN
        -- Insertar en tabla de notificaciones (si existe)
        INSERT INTO Notifications (userId, message, type, createdAt)
        VALUES (
            NEW.studentId, 
            CONCAT('Su solicitud de ', NEW.type, ' ha sido ', NEW.status),
            'accommodation_status',
            CURRENT_TIMESTAMP
        );
        
        -- Log del cambio
        INSERT INTO AuditLog (tableName, recordId, action, oldValue, newValue, userId, timestamp)
        VALUES (
            'Accommodations',
            NEW.id,
            'status_change',
            OLD.status,
            NEW.status,
            NEW.teacherId,
            CURRENT_TIMESTAMP
        );
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Aplicar el trigger
CREATE TRIGGER accommodation_status_notification
    AFTER UPDATE ON Accommodations
    FOR EACH ROW
    EXECUTE FUNCTION notify_accommodation_status_change();
```

### 3. Trigger para Validación de Límites de Solicitudes

```sql
-- Función para validar límites de solicitudes por estudiante
CREATE FUNCTION validate_student_request_limits()
RETURNS TRIGGER AS $$
DECLARE
    pending_count INTEGER;
BEGIN
    -- Contar solicitudes pendientes del estudiante
    SELECT COUNT(*) INTO pending_count
    FROM Accommodations 
    WHERE studentId = NEW.studentId AND status = 'pending';
    
    -- Límite máximo de 3 solicitudes pendientes
    IF pending_count >= 3 THEN
        RAISE EXCEPTION 'El estudiante ya tiene el máximo de solicitudes pendientes (3)';
    END IF;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Aplicar el trigger
CREATE TRIGGER check_student_request_limits
    BEFORE INSERT ON Accommodations
    FOR EACH ROW
    EXECUTE FUNCTION validate_student_request_limits();
```

### 4. Trigger para Auditoría de Cambios

```sql
-- Función para auditoría de cambios
CREATE FUNCTION audit_changes()
RETURNS TRIGGER AS $$
BEGIN
    -- Registrar cambios en tabla de auditoría
    INSERT INTO AuditLog (tableName, recordId, action, oldValue, newValue, userId, timestamp)
    VALUES (
        TG_TABLE_NAME,
        COALESCE(NEW.id, OLD.id),
        TG_OP,
        row_to_json(OLD),
        row_to_json(NEW),
        COALESCE(NEW.updatedBy, OLD.updatedBy),
        CURRENT_TIMESTAMP
    );
    
    RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

-- Aplicar triggers de auditoría
CREATE TRIGGER audit_users_changes
    AFTER INSERT OR UPDATE OR DELETE ON Users
    FOR EACH ROW
    EXECUTE FUNCTION audit_changes();

CREATE TRIGGER audit_accommodations_changes
    AFTER INSERT OR UPDATE OR DELETE ON Accommodations
    FOR EACH ROW
    EXECUTE FUNCTION audit_changes();
```

## Scripts de Inicialización

### 1. Script de Sincronización de Base de Datos

```javascript
// scripts/initDatabase.js
const { sequelize } = require('../src/models');

async function initializeDatabase() {
  try {
    // Sincronizar modelos con la base de datos
    await sequelize.sync({ force: false });
    console.log('✅ Base de datos sincronizada correctamente');
    
    // Crear índices adicionales
    await sequelize.query(`
      CREATE INDEX IF NOT EXISTS idx_accommodations_requested_date 
      ON Accommodations (requestedDate);
    `);
    
    await sequelize.query(`
      CREATE INDEX IF NOT EXISTS idx_absences_fecha_materia 
      ON Absences (fecha, materia);
    `);
    
    console.log('✅ Índices creados correctamente');
    
  } catch (error) {
    console.error('❌ Error al inicializar la base de datos:', error);
    throw error;
  }
}

module.exports = { initializeDatabase };
```
